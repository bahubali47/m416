3 no CREATE A USER-DIFINE EXCEPTION INVAID_AGE AND RAISE IT WHEN AN EMPLOYESS AGE IS OUTSIDE THE PERMITTED RANGE.


SET SERVEROUTPUT ON;

DECLARE
    INVALID_AGE EXCEPTION;
    V_AGE NUMBER;
BEGIN
    V_AGE := &AGE;

    IF V_AGE < 18 OR V_AGE > 60 THEN
        RAISE INVALID_AGE;
    ELSE
        DBMS_OUTPUT.PUT_LINE('Age is Valid');
    END IF;

EXCEPTION
    WHEN INVALID_AGE THEN
        DBMS_OUTPUT.PUT_LINE('Invalid Age');
END;
/
