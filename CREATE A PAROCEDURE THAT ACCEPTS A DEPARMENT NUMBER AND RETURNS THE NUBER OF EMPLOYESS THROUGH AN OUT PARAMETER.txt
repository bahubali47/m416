14. CREATE A PAROCEDURE THAT ACCEPTS A DEPARMENT NUMBER AND RETURNS THE NUBER OF EMPLOYESS THROUGH AN OUT PARAMETER.


Step 1: Create Procedure



CREATE OR REPLACE PROCEDURE COUNT_EMP
(
    P_DEPTNO IN NUMBER,
    P_TOTAL OUT NUMBER
)
IS
BEGIN
    SELECT COUNT(*)
    INTO P_TOTAL
    FROM EMP
    WHERE DEPTNO = P_DEPTNO;
END;
/


Step 2: Execute Procedure



SET SERVEROUTPUT ON;

DECLARE
    V_TOTAL NUMBER;
BEGIN
    COUNT_EMP(10, V_TOTAL);

    DBMS_OUTPUT.PUT_LINE('Total Employees: ' || V_TOTAL);
END;
/
