10. CREATE A SALARY_HISTORY TABLE AND A TRIGGER TO STORE EMPLOYEE ID, OLD SALARY, NEW SALARY, AND CHANGE DATE WHENEVER SALARY IS UPDATE.



Step 1: Create SALARY_HISTORY Table


CREATE TABLE SALARY_HISTORY
(
    EMPNO NUMBER,
    OLD_SAL NUMBER,
    NEW_SAL NUMBER,
    CHANGE_DATE DATE
);


Step 2: Create Trigger


CREATE OR REPLACE TRIGGER SALARY_HISTORY_TRG
AFTER UPDATE OF SAL ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO SALARY_HISTORY
    VALUES
    (
        :OLD.EMPNO,
        :OLD.SAL,
        :NEW.SAL,
        SYSDATE
    );
END;
/


Step 3: Update Employee Salary (Testing)


UPDATE EMP
SET SAL = 35000
WHERE EMPNO = 101;


Step 4: Check SALARY_HISTORY Table


SELECT * FROM SALARY_HISTORY;
