 1 no   WRITE A PL SQL THAT ACCEPTS AN EMPLOYEE ID AND DISPLAYS AN APPROPTIATE MESSAGE WHEN THE EMPLOYEE DOSE NOT EXITS



SET SERVEROUTPUT ON;

DECLARE
    V_ID EMP.EMPNO%TYPE;
    V_NAME EMP.ENAME%TYPE;

BEGIN
    V_ID := &EMPLOYEE_ID;

    SELECT ENAME INTO V_NAME
    FROM EMP
    WHERE EMPNO = V_ID;

    DBMS_OUTPUT.PUT_LINE('Employee Name: ' || V_NAME);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee Does Not Exist');

END;
/
