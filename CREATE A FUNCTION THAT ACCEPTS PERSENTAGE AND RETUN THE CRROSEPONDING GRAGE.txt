6. CREATE A FUNCTION THAT ACCEPTS PERSENTAGE AND RETUN THE CRROSEPONDING GRAGE.


CREATE OR REPLACE FUNCTION GET_GRADE
(P NUMBER)
RETURN VARCHAR2
IS
BEGIN
    IF P >= 90 THEN
        RETURN 'A+';
    ELSIF P >= 75 THEN
        RETURN 'A';
    ELSIF P >= 60 THEN
        RETURN 'B';
    ELSIF P >= 40 THEN
        RETURN 'C';
    ELSE
        RETURN 'FAIL';
    END IF;
END;
/


Execute Function:


SELECT GET_GRADE(85) FROM DUAL;
