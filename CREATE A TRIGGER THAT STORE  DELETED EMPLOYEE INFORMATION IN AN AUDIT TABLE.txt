9. CREATE A TRIGGER THAT STORE  DELETED EMPLOYEE INFORMATION IN AN AUDIT TABLE.


Step 1: Create EMP_DELETE_AUDIT Table
CREATE TABLE EMP_DELETE_AUDIT
(
    EMPNO NUMBER,
    ENAME VARCHAR2(50),
    SAL NUMBER,
    DEPTNO NUMBER,
    DELETE_DATE DATE
);


Step 2: Create Trigger


CREATE OR REPLACE TRIGGER EMP_DELETE_TRG
AFTER DELETE ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_DELETE_AUDIT
    VALUES
    (
        :OLD.EMPNO,
        :OLD.ENAME,
        :OLD.SAL,
        :OLD.DEPTNO,
        SYSDATE
    );
END;
/


Step 3: Delete Employee Record (Testing)


DELETE FROM EMP
WHERE EMPNO = 101;


Step 4: Check Audit Table


SELECT * FROM EMP_DELETE_AUDIT;
