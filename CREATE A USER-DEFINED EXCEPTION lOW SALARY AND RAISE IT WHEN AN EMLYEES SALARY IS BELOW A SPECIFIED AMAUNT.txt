2 no  CREATE A USER-DEFINED EXCEPTION lOW SALARY AND RAISE IT WHEN AN EMLYEES SALARY IS BELOW A SPECIFIED AMAUNT


SET SERVEROUTPUT ON;

DECLARE
    LOW_SALARY EXCEPTION;
    V_SAL NUMBER;
BEGIN
    V_SAL := &SALARY;

    IF V_SAL < 15000 THEN
        RAISE LOW_SALARY;
    ELSE
        DBMS_OUTPUT.PUT_LINE('Salary is Valid');
    END IF;

EXCEPTION
    WHEN LOW_SALARY THEN
        DBMS_OUTPUT.PUT_LINE('Salary is Too Low');
END;
/
