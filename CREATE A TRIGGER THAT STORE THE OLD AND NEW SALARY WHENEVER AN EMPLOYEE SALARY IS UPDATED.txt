8.  CREATE A TRIGGER THAT STORE THE OLD AND NEW SALARY WHENEVER AN EMPLOYEE SALARY IS UPDATED.


Step 1: Create SALARY_AUDIT Table


CREATE TABLE SALARY_AUDIT
(
    EMPNO NUMBER,
    OLD_SAL NUMBER,
    NEW_SAL NUMBER
);


Step 2: Create Trigger


CREATE OR REPLACE TRIGGER SAL_UPDATE_AUDIT
AFTER UPDATE OF SAL ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO SALARY_AUDIT
    VALUES
    (
        :OLD.EMPNO,
        :OLD.SAL,
        :NEW.SAL
    );
END;
/


Step 3: Update Employee Salary (Testing)


UPDATE EMP
SET SAL = 30000
WHERE EMPNO = 101;


Step 4: Check Audit Table


SELECT * FROM SALARY_AUDIT;
