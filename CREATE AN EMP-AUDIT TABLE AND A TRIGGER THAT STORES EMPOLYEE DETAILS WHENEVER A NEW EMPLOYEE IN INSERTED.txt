7. CREATE AN EMP-AUDIT TABLE AND A TRIGGER THAT STORES EMPOLYEE DETAILS WHENEVER A NEW EMPLOYEE IN INSERTED


Create EMP_AUDIT Table
CREATE TABLE EMP_AUDIT
(
    EMPNO NUMBER,
    ENAME VARCHAR2(50),
    SAL NUMBER,
    DEPTNO NUMBER,
    INSERT_DATE DATE
);


Step 2: Create Trigger

CREATE OR REPLACE TRIGGER EMP_INSERT_AUDIT
AFTER INSERT ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_AUDIT
    VALUES
    (
        :NEW.EMPNO,
        :NEW.ENAME,
        :NEW.SAL,
        :NEW.DEPTNO,
        SYSDATE
    );
END;
/
