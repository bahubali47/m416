5. CREATE A FUNCTION THAT ACCEPTS TWO NUMBERS AND RETURN THE GREATED NUMBER.

CREATE OR REPLACE FUNCTION GREATEST_NUM
(A NUMBER, B NUMBER)
RETURN NUMBER
IS
BEGIN
    IF A > B THEN
        RETURN A;
    ELSE
        RETURN B;
    END IF;
END;
/



Execute Function:


SELECT GREATEST_NUM(10, 20) FROM DUAL;
